structure{
  name: Model,
  type_parameters: [],
  parameters: [],
  fields: [
    {name: name, type: String},
    {name: kappa, type: String},
    {name: imports, type: (List String)},

  ]

definition{
  name: isOpen,
  parameters: [state : State],
  result: Bool,
  body: (match state
    | State.opened => true
    | State.observed => false
    | State.proved => false
    | State.refuted => false
    | State.retired => false)
}

definition{
  name: isAuthority,
  parameters: [role : Role],
  result: Bool,
  body: (match role
    | Role.authoritativeInput => true
    | Role.implementation => false
    | Role.projection => false
    | Role.externalDependency => false
    | Role.historical => false)
}

definition{
  name: authorities,
  parameters: [projections : (List Projection)],
  result: Nat,
  body: (match projections
    | List.nil => 0
    | List.cons head tail => (if (isAuthority (project head role)) then (1 + (authorities tail)) else (authorities tail))),
  axioms: [],
  recursive_argument: projections
}

definition{
  name: openCount,
  parameters: [models : (List Model)],
  result: Nat,
  body: (match models
    | List.nil => 0
    | List.cons head tail => (if (isOpen (project head state)) then (1 + (openCount tail)) else (openCount tail))),
  axioms: [],
  recursive_argument: models
}

definition{
  name: atMostOneAuthority,
  parameters: [models : (List Model)],
  result: Bool,
  body: (match models
    | List.nil => true
    | List.cons head tail => (if ((authorities (project head projections)) <=? 1) then (atMostOneAuthority tail) else false)),
  axioms: [],
  recursive_argument: models
}

definition{
  name: exactlyOneAuthority,
  parameters: [models : (List Model)],
  result: Bool,
  body: (match models
    | List.nil => true
    | List.cons head tail => (if ((authorities (project head projections)) =? 1) then (exactlyOneAuthority tail) else false)),
  axioms: [],
  recursive_argument: models
}
