definition{
  name: applicable,
  parameters: [characteristics : (List String), rules : (List Rule)],
  result: (List String),
  body: (match rules
    | List.nil => (nil String)
    | List.cons head tail => (if (Factory.Registry::memberString (project head characteristic) characteristics) then (cons (project head framework) (applicable characteristics tail)) else (applicable characteristics tail))),
  axioms: [],
  recursive_argument: rules
}

definition{
  name: obligationsOf,
  parameters: [ids : (List String), frameworks : (List Framework)],
  result: (List String),
  body: (match frameworks
    | List.nil => (nil String)
    | List.cons head tail => (if (Factory.Registry::memberString (project head id) ids) then (prim append (project head obligations) (obligationsOf ids tail) : (List String)) else (obligationsOf ids tail))),
  axioms: [],
  recursive_argument: frameworks
}

definition{
  name: compliant,
  parameters: [concept : Concept, rules : (List Rule), frameworks : (List Framework)],
  result: Bool,
  body: (match (Factory.Registry::unreceipted (obligationsOf (applicable (project concept characteristics) rules) frameworks) (project concept receipts))
    | List.nil => true
    | List.cons head tail => false)
}
