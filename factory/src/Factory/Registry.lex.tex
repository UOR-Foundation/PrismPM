definition{
  name: sameVersion,
  parameters: [left : Version, right : Version],
  result: Bool,
  body: (((project left major) =? (project right major)) && (((project left minor) =? (project right minor)) && ((project left patch) =? (project right patch))))
}

definition{
  name: next,
  parameters: [version : Version, level : Nat],
  result: Version,
  body: (record Version {major := (if (level =? 3) then ((project version major) + 1) else (project version major)), minor := (if (level =? 3) then 0 else (if (level =? 2) then ((project version minor) + 1) else (project version minor))), patch := (if (level =? 3) then 0 else (if (level =? 2) then 0 else (if (level =? 1) then ((project version patch) + 1) else (project version patch))))})
}

definition{
  name: later,
  parameters: [left : Version, right : Version],
  result: Bool,
  body: (if ((project left major) =? (project right major)) then (if ((project left minor) =? (project right minor)) then ((project right patch) <? (project left patch)) else ((project right minor) <? (project left minor))) else ((project right major) <? (project left major)))
}

definition{
  name: memberString,
  parameters: [value : String, values : (List String)],
  result: Bool,
  body: (match values
    | List.nil => false
    | List.cons head tail => (if (prim equal value head : Bool) then true else (memberString value tail))),
  axioms: [],
  recursive_argument: values
}

definition{
  name: unreceipted,
  parameters: [obligations : (List String), receipts : (List String)],
  result: (List String),
  body: (match obligations
    | List.nil => (nil String)
    | List.cons head tail => (if (memberString head receipts) then (unreceipted tail receipts) else (cons head (unreceipted tail receipts)))),
  axioms: [],
  recursive_argument: obligations
}

definition{
  name: resolves,
  parameters: [imports : (List String), accepted : (List String)],

  body: (match imports
    | List.nil => true
    | List.cons head tail => (if (memberString head accepted) then (resolves tail accepted) else false)),
  axioms: [],
  recursive_argument: imports
}

definition{
  name: route,
  parameters: [imports : (List String), accepted : (List String), obligations : (List String), receipts : (List String)],
  result: String,
  body: (if (resolves imports accepted) then (match (unreceipted obligations receipts)
    | List.nil => "proved"
    | List.cons head tail => "observed") else "refuted")
}

definition{

    | List.nil => true
    | List.cons head tail => (if (bound <? head) then (increasing head tail) else false)),
  axioms: [],
  recursive_argument: values
}

definition{
  name: mergeSorted,
  parameters: [fuel : Nat, left : (List Nat), right : (List Nat)],
  result: (List Nat),
  body: (match fuel
    | Nat.zero => (prim append left right : (List Nat))
    | Nat.succ smaller => (match left
      | List.nil => right
      | List.cons l ls => (match right
        | List.nil => left
        | List.cons r rs => (if (l <? r) then (cons l (mergeSorted smaller ls right)) else (cons r (mergeSorted smaller left rs)))))),
  axioms: [],
  recursive_argument: fuel
}

definition{
  name: mergeAll,
  parameters: [left : (List Nat), right : (List Nat)],
  result: (List Nat),
  body: (mergeSorted ((prim length left : Nat) + (prim length right : Nat)) left right)
}

definition{
  name: dedupe,
  parameters: [last : Nat, values : (List Nat)],
  result: (List Nat),
  body: (match values
    | List.nil => (nil Nat)
    | List.cons head tail => (if (head =? last) then (dedupe last tail) else (cons head (dedupe head tail)))),
  axioms: [],
  recursive_argument: values
}

definition{
  name: sameNats,
  parameters: [left : (List Nat), right : (List Nat)],
  result: Bool,
  body: (match left
    | List.nil => (match right
      | List.nil => true
      | List.cons r rs => false)
    | List.cons l ls => (match right
      | List.nil => false
      | List.cons r rs => (if (l =? r) then (sameNats ls rs) else false))),
  axioms: [],
  recursive_argument: left
}

definition{
  name: within,
  parameters: [fuel : Nat, small : (List Nat), big : (List Nat)],
  result: Bool,
  body: (match fuel
    | Nat.zero => false
    | Nat.succ smaller => (match small
      | List.nil => true
      | List.cons s ss => (match big
        | List.nil => false
        | List.cons b bs => (if (s =? b) then (within smaller ss bs) else (if (b <? s) then (within smaller small bs) else false))))),
  axioms: [],
  recursive_argument: fuel
}

definition{
  name: kappas,
  parameters: [items : (List Feature)],
  result: (List Nat),
  body: (match items
    | List.nil => (nil Nat)

  axioms: [],
  recursive_argument: items
}

definition{
  name: missing,
  parameters: [fuel : Nat, left : (List Feature), right : (List Feature)],
  result: (List Feature),
  body: (match fuel
    | Nat.zero => left
    | Nat.succ smaller => (match left
      | List.nil => (nil Feature)
      | List.cons l ls => (match right
        | List.nil => left
        | List.cons r rs => (if ((project l kappa) =? (project r kappa)) then (missing smaller ls rs) else (if ((project l kappa) <? (project r kappa)) then (cons l (missing smaller ls right)) else (missing smaller left rs)))))),
  axioms: [],
  recursive_argument: fuel
}

definition{
  name: largest,
  parameters: [left : Nat, right : Nat],
  result: Nat,
  body: (if (left <? right) then right else left)
}

definition{
  name: addedLevel,
  parameters: [item : Feature, dropped : (List Feature)],
  result: Nat,
  body: (match dropped
    | List.nil => 2
    | List.cons head tail => (if ((project head key) =? (project item key)) then (if (not ((project head interface) =? (project item interface))) then 3 else (if ((project head executable) && (not (project item executable))) then 3 else (if ((project item executable) && (not (project head executable))) then 2 else 1))) else (addedLevel item tail))),
  axioms: [],
  recursive_argument: dropped
}

definition{
  name: additionsLevel,
  parameters: [added : (List Feature), dropped : (List Feature)],
  result: Nat,
  body: (match added
    | List.nil => 0
    | List.cons head tail => (largest (addedLevel head dropped) (additionsLevel tail dropped))),
  axioms: [],
  recursive_argument: added
}

definition{
  name: keyHeld,

  body: (match items

  axioms: [],
  recursive_argument: items
}

definition{
  name: anyRemoved,
  parameters: [dropped : (List Feature), added : (List Feature)],
  result: Bool,
  body: (match dropped
    | List.nil => false
    | List.cons head tail => (if (keyHeld (project head key) added) then (anyRemoved tail added) else true)),
  axioms: [],
  recursive_argument: dropped
}

definition{
  name: levelOf,
  parameters: [added : (List Feature), dropped : (List Feature)],
  result: Nat,
  body: (if (anyRemoved dropped added) then 3 else (additionsLevel added dropped))
}

definition{
  name: changeLevel,
  parameters: [before : (List Feature), after : (List Feature)],
  result: Nat,
  body: (levelOf (missing ((prim length before : Nat) + (prim length after : Nat)) after before) (missing ((prim length before : Nat) + (prim length after : Nat)) before after))
}

definition{

(changeLevel (project (releaseAt (project current base) earlier) features) (project current features))) && (sameVersion (project current version) (next (highest earlier (noVersion)) (changeLevel (project (releaseAt (project current base) earlier) features) (project current features)))))))

definition{
  name: placementHolds,
  parameters: [placed : Placement, items : (List Component)],
  result: Bool,
  body: (match (project placed outcome)
    | Outcome.resolved => (sameNats (kappas (project (releaseOf (project placed version) (project (componentAt (project placed component) items) releases)) features)) (project placed features))
    | Outcome.contained => (within ((prim length (project placed features) : Nat) + (prim length (project (releaseOf (project placed version) (project (componentAt (project placed component) items) releases)) features) : Nat)) (project placed features) (kappas (project (releaseOf (project placed version) (project (componentAt (project placed component) items) releases)) features)))
    | Outcome.merged => (sameNats (kappas (project (releaseOf (project placed version) (project (componentAt (project placed component) items) releases)) features)) (dedupe 0 (mergeAll (kappas (project (releaseAt (project (releaseOf (project placed version) (project (componentAt (project placed component) items) releases)) base) (project (componentAt (project placed component) items) releases)) features)) (project placed features)))))
}
