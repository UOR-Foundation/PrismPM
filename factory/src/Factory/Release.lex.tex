definition{
  name: stageIndex,
  parameters: [stage : Stage],

    | Stage.installedVv => 5
    | Stage.publication => 6)

definition{

    | Stage.reproducibility => (cons "image-reproduced" (cons "browser-closure" (cons "library-closure" (nil String))))

    | Stage.installedSdk => (cons "product-built" (nil String))
    | Stage.installedVv => (cons "source-verified" (nil String))
    | Stage.publication => (cons "published" (nil String)))
}

definition{
  name: onProjection,
  parameters: [stage : Stage],
  result: Bool,
  body: (match stage
    | Stage.sourceVv => false
    | Stage.images => true
    | Stage.reproducibility => true
    | Stage.native => true
    | Stage.installedSdk => true
    | Stage.installedVv => true
    | Stage.publication => false)
}

definition{
  name: waitsForTheSource,
  parameters: [stage : Stage],
  result: Bool,
  body: (match stage
    | Stage.sourceVv => false
    | Stage.images => false
    | Stage.reproducibility => false
    | Stage.native => false
    | Stage.installedSdk => false
    | Stage.installedVv => true
    | Stage.publication => true)
}

definition{
  name: occurrences,
  parameters: [obligation : String, stages : (List Stage)],
  result: Nat,
  body: (match stages
    | List.nil => 0
    | List.cons head tail => (if (Factory.Registry::memberString obligation (establishes head)) then (1 + (occurrences obligation tail)) else (occurrences obligation tail))),
  axioms: [],
  recursive_argument: stages
}

definition{
  name: eachOnce,
  parameters: [wanted : (List String), stages : (List Stage)],
  result: Bool,
  body: (match wanted
    | List.nil => true
    | List.cons head tail => (((occurrences head stages) =? 1) && (eachOnce tail stages))),
  axioms: [],
  recursive_argument: wanted
}

definition{
  name: projectionsReestablishTheSource,
  parameters: [stages : (List Stage)],
  result: Nat,
  body: (match stages
    | List.nil => 0
    | List.cons head tail => (if ((onProjection head) && (Factory.Registry::memberString "source-verified" (establishes head))) then (1 + (projectionsReestablishTheSource tail)) else (projectionsReestablishTheSource tail))),
  axioms: [],
  recursive_argument: stages
}

definition{
  name: waiting,
  parameters: [stages : (List Stage)],
  result: Nat,
  body: (match stages
    | List.nil => 0
    | List.cons head tail => (if (waitsForTheSource head) then (1 + (waiting tail)) else (waiting tail))),
  axioms: [],
  recursive_argument: stages
}

definition{
  name: imageBuilds,
  parameters: [stage : Stage],
  result: Nat,
  body: (match stage
    | Stage.sourceVv => 0
    | Stage.images => 1
    | Stage.reproducibility => 1
    | Stage.native => 0
    | Stage.installedSdk => 0
    | Stage.installedVv => 0
    | Stage.publication => 0)
}

definition{
  name: builds,
  parameters: [stages : (List Stage)],
  result: Nat,
  body: (match stages
    | List.nil => 0
    | List.cons head tail => ((imageBuilds head) + (builds tail))),
  axioms: [],
  recursive_argument: stages
}

definition{
  name: route,
  parameters: [],
  result: (List Stage),

}

  name: priorRoute,

}

definition{
  name: sourceRoute,

definition{
