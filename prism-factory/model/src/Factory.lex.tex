\begin{lexlean}{Factory}
\useglossary{lexlean.std.nat@1.2.0}
\importmodule{LexLeanStd.Text}
\importmodule{Factory.Engine}
\importmodule{Factory.PrismPM}
\importmodule{Factory.Workflow}
\title{Natural number addition}
\begin{semanticmodule}
structure{
  name: Request,
  type_parameters: [],
  parameters: [],
  fields: [
    {name: tree, type: (List (Bytes * Bytes))},
    {name: receipts, type: (List (String * Bool))},
    {name: weights, type: (List (String * String))},
    {name: outcomes, type: (List (String * Bool))}
  ]
}

definition{
  name: cut,
  parameters: [raw : Bytes],
  result: (Bytes * Bytes),
  body: (match (LexLeanStd.Text::find raw 0x20)
    | Option.some position => ((LexLeanStd.Text::sliceOr raw 0 position), (LexLeanStd.Text::sliceOr raw (position + 1) (prim subtract (prim length raw : Nat) (position + 1) : Nat)))
    | Option.none => (raw, 0x)),
  executable: true
}

definition{
  name: passes,
  parameters: [raw : Bytes],
  result: Bool,
  body: (prim equal raw (Factory.Engine::text "pass") : Bool),
  executable: true
}

definition{
  name: readLine,
  parameters: [request : Request, raw : Bytes],
  result: Request,
  body: (let split : (Bytes * Bytes) := (cut raw); (let tag : Bytes := (first split); (let rest : (Bytes * Bytes) := (cut (second split)); (if (prim equal tag (Factory.Engine::text "tree") : Bool) then (record Request {tree := (cons ((first rest), (second rest)) (project request tree)), receipts := (project request receipts), weights := (project request weights), outcomes := (project request outcomes)}) else (if (prim equal tag (Factory.Engine::text "receipt") : Bool) then (record Request {tree := (project request tree), receipts := (cons ((Factory.Engine::decoded (first rest)), (passes (second rest))) (project request receipts)), weights := (project request weights), outcomes := (project request outcomes)}) else (if (prim equal tag (Factory.Engine::text "weight") : Bool) then (record Request {tree := (project request tree), receipts := (project request receipts), weights := (cons ((Factory.Engine::decoded (first rest)), (Factory.Engine::decoded (second rest))) (project request weights)), outcomes := (project request outcomes)}) else (if (prim equal tag (Factory.Engine::text "outcome") : Bool) then (record Request {tree := (project request tree), receipts := (project request receipts), weights := (project request weights), outcomes := (cons ((Factory.Engine::decoded (first rest)), (passes (second rest))) (project request outcomes))}) else request))))))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: reversedTree,
  parameters: [tree : (List (Bytes * Bytes))],
  result: (List (Bytes * Bytes)),
  body: (prim list_fold (fun (sofar : (List (Bytes * Bytes))) (entry : (Bytes * Bytes)) => (cons entry sofar)) (nil (Bytes * Bytes)) tree : (List (Bytes * Bytes))),
  executable: true
}

definition{
  name: reversedPairs,
  parameters: [pairs : (List (String * String))],
  result: (List (String * String)),
  body: (prim list_fold (fun (sofar : (List (String * String))) (entry : (String * String)) => (cons entry sofar)) (nil (String * String)) pairs : (List (String * String))),
  executable: true
}

definition{
  name: readRequest,
  parameters: [raw : Bytes],
  result: Request,
  body: (let parsed : Request := (prim list_fold (fun (request : Request) (piece : Bytes) => (readLine request piece)) (record Request {tree := (nil (Bytes * Bytes)), receipts := (nil (String * Bool)), weights := (nil (String * String)), outcomes := (nil (String * Bool))}) (LexLeanStd.Text::lines raw) : Request); (record Request {tree := (reversedTree (project parsed tree)), receipts := (project parsed receipts), weights := (reversedPairs (project parsed weights)), outcomes := (project parsed outcomes)})),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: levelOf,
  parameters: [wanted : String, weights : (List (String * String))],
  result: Factory.Engine::Level,
  body: (Factory.Engine::levelNamed (prim list_fold (fun [wanted] (sofar : String) (entry : (String * String)) => (if (prim equal (first entry) wanted : Bool) then (second entry) else sofar)) "4" weights : String)),
  executable: true
}

definition{
  name: insertByLevel,
  parameters: [item : (String * Nat), sorted : (List (String * Nat))],
  result: (List (String * Nat)),
  body: (match sorted
    | List.nil => (cons item (nil (String * Nat)))
    | List.cons head tail => (if ((second head) <? (second item)) then (cons item (cons head tail)) else (cons head (insertByLevel item tail)))),
  executable: true,
  recursive_argument: sorted
}

definition{
  name: schedule,
  parameters: [planned : (List Factory.Engine::Planned), weights : (List (String * String))],
  result: (List (String * Nat)),
  body: (prim list_fold (fun [weights] (sorted : (List (String * Nat))) (entry : Factory.Engine::Planned) => (if (Factory.Engine::isObserved (project entry verdict)) then (insertByLevel ((project (project entry obligation) name), (Factory.Engine::levelValue (levelOf (project (project entry obligation) name) weights))) sorted) else sorted)) (nil (String * Nat)) planned : (List (String * Nat))),
  executable: true
}

definition{
  name: words,
  parameters: [pieces : (List String)],
  result: Bytes,
  body: (Factory.Engine::line (prim list_fold (fun (sofar : (List Bytes)) (piece : String) => (prim append sofar (cons (Factory.Engine::text piece) (nil Bytes)) : (List Bytes))) (nil Bytes) pieces : (List Bytes))),
  executable: true
}

definition{
  name: tagged,
  parameters: [tag : String, owner : String, values : (List String)],
  result: Bytes,
  body: (Factory.Engine::joinedReversed (prim list_fold (fun [owner, tag] (sofar : (List Bytes)) (value : String) => (cons (words (cons tag (cons owner (cons value (nil String))))) sofar)) (nil Bytes) values : (List Bytes))),
  executable: true
}

definition{
  name: closureFiles,
  parameters: [owner : String, inputs : (List String), tree : (List (Bytes * Bytes))],
  result: Bytes,
  body: (Factory.Engine::joinedReversed (prim list_fold (fun [inputs, owner] (sofar : (List Bytes)) (entry : (Bytes * Bytes)) => (if (Factory.Engine::covered inputs (second entry)) then (cons (Factory.Engine::line (cons (Factory.Engine::text "file") (cons (Factory.Engine::text owner) (cons (second entry) (nil Bytes))))) sofar) else sofar)) (nil Bytes) tree : (List Bytes))),
  executable: true
}

definition{
  name: runsHere,
  parameters: [entry : Factory.Engine::Planned],
  result: Bool,
  body: ((Factory.Engine::isObserved (project entry verdict)) && (prim equal (project (project entry obligation) environment) "host" : Bool)),
  executable: true
}

definition{
  name: describe,
  parameters: [entry : Factory.Engine::Planned, tree : (List (Bytes * Bytes))],
  result: Bytes,
  body: (let item : Factory.Engine::Obligation := (project entry obligation); (let owner : String := (project item name); (let head : Bytes := (words (cons "obligation" (cons owner (cons (Factory.Engine::routeName (project entry verdict)) (cons (project entry kappa) (nil String)))))); (let body : Bytes := (prim append (prim append (words (cons "kind" (cons owner (cons (Factory.Engine::kindName (project item kind)) (nil String))))) (words (cons "environment" (cons owner (cons (project item environment) (nil String))))) : Bytes) (words (cons "run" (cons owner (cons (project item run) (nil String))))) : Bytes); (let files : Bytes := (if (runsHere entry) then (closureFiles owner (project item inputs) tree) else 0x); (prim append (prim append head body : Bytes) (prim append (prim append (tagged "input" owner (project item inputs)) (tagged "needs" owner (project item needs)) : Bytes) files : Bytes) : Bytes)))))),
  executable: true
}

definition{
  name: render,
  parameters: [planned : (List Factory.Engine::Planned), weights : (List (String * String)), tree : (List (Bytes * Bytes))],
  result: Bytes,
  body: (let described : Bytes := (Factory.Engine::joinedReversed (prim list_fold (fun [tree] (sofar : (List Bytes)) (entry : Factory.Engine::Planned) => (cons (describe entry tree) sofar)) (nil Bytes) planned : (List Bytes))); (let ordered : Bytes := (prim list_fold (fun (sofar : Bytes) (entry : (String * Nat)) => (prim append sofar (words (cons "schedule" (cons (first entry) (nil String)))) : Bytes)) 0x (schedule planned weights) : Bytes); (prim append (prim append (prim append described ordered : Bytes) (words (cons "cuttable" (cons (if (Factory.Engine::cuttable planned) then "yes" else "no") (nil String)))) : Bytes) (words (cons "releasable" (cons (if (Factory.Engine::releasable planned) then "yes" else "no") (nil String)))) : Bytes))),
  executable: true
}

definition{
  name: plan,
  parameters: [request : Bytes],
  result: Bytes,
  body: (let parsed : Request := (readRequest request); (render (Factory.Engine::planAll (Factory.PrismPM::graph) (project parsed tree) (project parsed receipts)) (project parsed weights) (project parsed tree))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true,
  production: {effects: [allocation, overflow, recursion], targets: [rust-std]}
}

definition{
  name: outcomeOf,
  parameters: [wanted : String, outcomes : (List (String * Bool))],
  result: (Option Bool),
  body: (prim list_fold (fun [wanted] (sofar : (Option Bool)) (entry : (String * Bool)) => (if (prim equal (first entry) wanted : Bool) then (ctor Option.some [Bool] (second entry)) else sofar)) (ctor Option.none [Bool]) outcomes : (Option Bool)),
  executable: true
}

definition{
  name: reweigh,
  parameters: [request : Bytes],
  result: Bytes,
  body: (let parsed : Request := (readRequest request); (prim list_fold (fun [parsed] (sofar : Bytes) (item : Factory.Engine::Obligation) => (let level : Factory.Engine::Level := (levelOf (project item name) (project parsed weights)); (let next : Factory.Engine::Level := (match (outcomeOf (project item name) (project parsed outcomes))
      | Option.some passed => (Factory.Engine::reinforce level passed)
      | Option.none => level); (prim append sofar (words (cons "weight" (cons (project item name) (cons (Factory.Engine::levelText next) (nil String))))) : Bytes)))) 0x (Factory.PrismPM::graph) : Bytes)),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true,
  production: {effects: [allocation, overflow, recursion], targets: [rust-std]}
}

definition{
  name: marker,
  parameters: [fragment : String],
  result: Bytes,
  body: (Factory.Engine::line (cons (Factory.Engine::text "#fragment") (cons (Factory.Engine::text fragment) (nil Bytes)))),
  executable: true
}

definition{
  name: fragmentLines,
  parameters: [names : (List String)],
  result: Bytes,
  body: (prim list_fold (fun (sofar : Bytes) (fragment : String) => (let marked : Bytes := (prim append sofar (marker fragment) : Bytes); (if (prim equal fragment "policy" : Bool) then (Factory.Workflow::render marked (Factory.Workflow::factoryJob)) else marked))) 0x names : Bytes),
  executable: true
}

definition{
  name: needsLine,
  parameters: [names : (List String)],
  result: Bytes,
  body: (prim append (prim append (Factory.Engine::text "    needs: [") (LexLeanStd.Text::joinWith (prim list_fold (fun (sofar : (List Bytes)) (fragment : String) => (prim append sofar (cons (Factory.Engine::text fragment) (nil Bytes)) : (List Bytes))) (nil Bytes) names : (List Bytes)) 0x2c20) : Bytes) (Factory.Engine::text "]\n") : Bytes),
  executable: true
}

definition{
  name: resultOf,
  parameters: [job : String],
  result: Bytes,
  body: (prim append (prim append (Factory.Engine::text "${{ needs.") (Factory.Engine::text job) : Bytes) (Factory.Engine::text ".result }}") : Bytes),
  executable: true
}

definition{
  name: resultLines,
  parameters: [table : (List (String * (List String)))],
  result: Bytes,
  body: (prim list_fold (fun (sofar : Bytes) (entry : (String * (List String))) => (prim append sofar (prim append (Factory.Engine::text "            ") (Factory.Engine::line (cons (Factory.Engine::text (first entry)) (prim list_fold (fun (pieces : (List Bytes)) (job : String) => (prim append pieces (cons (resultOf job) (nil Bytes)) : (List Bytes))) (nil Bytes) (second entry) : (List Bytes)))) : Bytes) : Bytes)) 0x table : Bytes),
  executable: true
}

definition{
  name: releaseWorkflow,
  parameters: [],
  result: Bytes,
  body: (let head : Bytes := (prim append (Factory.Workflow::render 0x (Factory.Workflow::releaseHead)) (marker "header") : Bytes); (let body : Bytes := (prim append head (fragmentLines (Factory.PrismPM::fragments)) : Bytes); (let receipts : Bytes := (prim append (Factory.Workflow::render body (Factory.Workflow::receiptsHead)) (needsLine (cons "factory" (Factory.PrismPM::fragments))) : Bytes); (Factory.Workflow::render (prim append (Factory.Workflow::render receipts (Factory.Workflow::receiptsBody)) (resultLines (Factory.PrismPM::jobs)) : Bytes) (Factory.Workflow::receiptsTail))))),
  executable: true
}

definition{
  name: workflow,
  parameters: [selector : Bytes],
  result: Bytes,
  body: (if (prim equal selector (Factory.Engine::text "release") : Bool) then (releaseWorkflow) else (if (prim equal selector (Factory.Engine::text "cut") : Bool) then (Factory.Workflow::render 0x (Factory.Workflow::cutLines)) else (Factory.Workflow::render 0x (Factory.Workflow::factoryLines)))),
  executable: true,
  production: {effects: [allocation, overflow, recursion], targets: [rust-std]}
}
