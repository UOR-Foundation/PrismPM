definition{
  name: swapCaseSearch,
  parameters: [octet : Bytes, from : Bytes, to : Bytes, count : Nat],
  result: Bytes,
  body: (match count
    | Nat.zero => octet
    | Nat.succ rest => (if (bytesEqual (piece from rest 1) octet) then (piece to rest 1) else (swapCaseSearch octet from to rest))),
  axioms: [propext],
  executable: true,
  recursive_argument: count
}

definition{
  name: upperOctet,

  executable: true

definition{
  name: camelFrom,

  body: (match remaining

    | Nat.succ rest => (if (isSeparator (piece source index 1)) then (camelFrom source (index + 1) true rest) else (prim append (if raise then (upperOctet (piece source index 1)) else (piece source index 1)) (camelFrom source (index + 1) false rest) : Bytes))),
  axioms: [propext],
  executable: true,
  recursive_argument: remaining
}

  axioms: [propext],
  executable: true,
  recursive_argument: remaining
}

definition{

  result: Bytes,
  body: (match remaining
    | Nat.zero => 0x
    | Nat.succ rest => (let octet : Bytes := (piece source index 1); (if (bytesEqual octet 0x2d) then (prim append 0x5f (snakeFrom source (index + 1) false rest) : Bytes) else (if (isUpperOctet octet) then (prim append (if after then (prim append 0x5f (lowerOctet octet) : Bytes) else (lowerOctet octet)) (snakeFrom source (index + 1) false rest) : Bytes) else (prim append octet (snakeFrom source (index + 1) (isLowerOrDigit octet) rest) : Bytes))))),
  axioms: [propext],
  executable: true,
  recursive_argument: remaining
}
