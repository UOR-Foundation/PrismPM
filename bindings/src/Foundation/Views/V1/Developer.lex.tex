definition{
  name: termOf,
  parameters: [name : String, kind : String, terms : (List DeveloperTerm)],
  result: (Option DeveloperTerm),
  body: (match terms
    | List.nil => (ctor Option.none [DeveloperTerm])
    | List.cons head rest => (if ((same (project head view) name) && (same (project head kind) kind)) then (ctor Option.some [DeveloperTerm] head) else (termOf name kind rest))),
  axioms: [],
  executable: true,
  recursive_argument: terms

definition{
  name: wordRefusal,
  parameters: [entry : DeveloperName],
  result: (List String),
  body: (if (same (project entry kind) "flag") then (nil String) else (match (termOf (project entry name) (if (same (project entry kind) "namespace") then "noun" else "verb") (developerTerms))
    | Option.some found => (if (same (project found since) "now") then (nil String) else (one (Foundation.Bindings.V1.Text::concat (cons (project entry place) (cons " takes a Docker word before Prism has its behaviour, which " (cons (project found since) (cons " builds" (nil String))))))))
    | Option.none => (if (Foundation.Bindings.V1.Text::memberText (project entry name) (developerWords)) then (nil String) else (one (Foundation.Bindings.V1.Text::concat (cons (project entry place) (cons " is not a word of the developer view; declare it in the glossary or rename it" (nil String)))))))),
  axioms: [propext],
  executable: true
}

definition{
  name: canonicalRefusal,
  parameters: [entry : DeveloperName],
  result: (List String),
  body: (match (canonicalIn (segments (project entry name)))
    | Option.none => (nil String)
    | Option.some found => (one (Foundation.Bindings.V1.Text::concat (cons (project entry place) (cons " uses the canonical term " (cons (first found) (cons ", which the developer view calls " (cons (second found) (nil String))))))))),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

  name: sameAlias,
  parameters: [left : DeveloperAlias, right : DeveloperAlias],
  result: Bool,

  axioms: [],
  executable: true

definition{
  name: hasAlias,
  parameters: [entry : DeveloperAlias, aliases : (List DeveloperAlias)],
  result: Bool,
  body: (match aliases
    | List.nil => false
    | List.cons head rest => ((sameAlias entry head) || (hasAlias entry rest))),
  axioms: [],
  executable: true,
  recursive_argument: aliases
}

  name: hasName,

  result: Bool,
  body: (match names
    | List.nil => false
    | List.cons head rest => (((same (project head kind) kind) && (same (project head name) name)) || (hasName kind name rest))),
  axioms: [],
  executable: true,
  recursive_argument: names
